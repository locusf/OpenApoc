#!/usr/bin/env python3
import argparse
import xml.etree.ElementTree as ET


def read_state(path):
    tree = ET.parse(path)
    root = tree.getroot()
    state = {"objects": [], "enums": []}
    if root.tag != "openapoc_gamestate":
        return state
    for node in root:
        if node.tag == "object":
            obj = {
                "name": "",
                "external": node.attrib.get("external") == "true",
                "full": node.attrib.get("full") == "true",
                "members": [],
            }
            for child in node:
                if child.tag == "member":
                    obj["members"].append((child.text or "", child.attrib.get("type", "Normal")))
                elif child.tag == "name":
                    obj["name"] = child.text or ""
            state["objects"].append(obj)
        elif node.tag == "enum":
            enum = {
                "name": "",
                "external": node.attrib.get("external") == "true",
                "values": [],
            }
            for child in node:
                if child.tag == "value":
                    enum["values"].append(child.text or "")
                elif child.tag == "name":
                    enum["name"] = child.text or ""
            state["enums"].append(enum)
    return state


def source_member_fns(kind):
    if kind == "Normal":
        return "serializeIn", "getNode", "serializeOut", "addNode"
    if kind == "Section":
        return "serializeIn", "getSection", "serializeOut", "addSection"
    if kind == "SectionMap":
        return "serializeInSectionMap", "getSection", "serializeOutSectionMap", "addSection"
    raise ValueError(kind)

                out.write(f"\tif (a.{member_name} != b.{member_name}) {{ return false;}}\n")
            out.write("\treturn true;\n}\n")
            if not obj["external"]:
                out.write("inline\n")
            out.write(f"bool operator!=(const {name} &a, const {name} &b)\n{{\n\treturn !(a == b);\n}}\n")
        for enum in state["enums"]:
            name = enum["name"]

                out.write("inline\n")
            out.write(f"void serializeIn(const GameState *state, SerializationNode* node, {name} &val)\n{{\n")
            out.write(f"\tstatic const std::map<{name}, UString> valueMap = {{\n")
            for value in enum["values"]:
                out.write(f"\t\t{{{name}::{value}, \"{value}\"}},\n")
            out.write("\t};\n")
            out.write("\tserializeIn(state, node, val, valueMap);\n}\n")
            if not enum["external"]:
                out.write("inline\n")
            out.write(f"void serializeOut(SerializationNode* node, const {name} &val, const {name} &ref)\n{{\n")
            out.write(f"\tstatic const std::map<{name}, UString> valueMap = {{\n")
            for value in enum["values"]:
                out.write(f"\t\t{{{name}::{value}, \"{value}\"}},\n")
            out.write("\t};\n")
            out.write("\tserializeOut(node, val, ref, valueMap);\n}\n")
        out.write("\n} // namespace OpenApoc\n")


def main():
    ap = argparse.ArgumentParser(add_help=False)
    ap.add_argument("--help", action="store_true")
    ap.add_argument("-x", "--xml")
    ap.add_argument("-h", "--output-header")
    ap.add_argument("-o", "--output-source")
    ns = ap.parse_args()
    if ns.help or not ns.xml or not ns.output_header or not ns.output_source:
        raise SystemExit(1)
    state = read_state(ns.xml)
    write_header(ns.output_header, state)
    write_source(ns.output_source, state)


if __name__ == "__main__":
    main()

import xml.etree.ElementTree as ET

    tree = ET.parse(path)
    root = tree.getroot()

def split_ns(name):
    return name.split("::") if name else [""]


def iterate_depth_first(names, out, process, element_size):
    previous = []
    for e in names:
        current = split_ns(e["name"])
        i = 0
        while i < len(previous) and i < len(current) and previous[i] == current[i]:
            i += 1
        if previous:
            for part in reversed(previous[i+1:]):
                out.write(f"\tlua_setfield(L, -2, \"{part}\");\n")
            out.write(f"\tlua_setfield(L, -2, \"{previous[i]}\");\n")
        for part in current[i:-1]:
            out.write("\tlua_createtable(L, 0, 1);\n")
        a, b = element_size(e)
        out.write(f"\tlua_createtable(L, {a}, {b});\n")
        process(e)
        previous = current
    for part in reversed(previous):
        out.write(f"\tlua_setfield(L, -2, \"{part}\");\n")

    with open(path, "w", encoding="utf-8") as out:

        out.write("extern \"C\"\n{\n")

        out.write("}\n\n")
        out.write("namespace OpenApoc {\n\n")
        out.write("class SerializationNode;\n\n")

            out.write(f"void pushToLua(lua_State *L, {name} &obj);\n")
            out.write(f"void pushToLua(lua_State *L, const {name} &obj);\n")
        for enum in state["enums"]:
            if not enum["external"]:
                continue
            out.write(f"void pushToLua(lua_State *L, {enum['name']} val);\n")
        out.write("\n} // namespace OpenApoc\n")
