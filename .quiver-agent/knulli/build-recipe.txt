make_copygen() {
  local src_h="$1"
  local src_cpp="$2"
  local out="$3"
  cat > "$out" <<EOF
#!/bin/sh
set -eu
SRC_H='$src_h'
SRC_CPP='$src_cpp'
OUT_H=
OUT_CPP=
while [ "\$#" -gt 0 ]; do
  case "\$1" in
    -x)
      shift 2
      ;;
    -h)
      OUT_H="\$2"
      shift 2
      ;;
    -o)
      OUT_CPP="\$2"
      shift 2
      ;;
    *)
      shift
      ;;
  esac
done
[ -n "\${OUT_H}" ]
[ -n "\${OUT_CPP}" ]
cp "\$SRC_H" "\$OUT_H"
cp "\$SRC_CPP" "\$OUT_CPP"
EOF
  chmod +x "$out"
}

build_native_gen() {
  local compiler="$1"
  local main_src="$2"
  local out_bin="$3"
  "$compiler" -x c++ -std=c++17 \
    -I. -Itools/code_generators -Idependencies/pugixml/src \
    "$main_src" tools/code_generators/serialization_xml.cpp dependencies/pugixml/src/pugixml.cpp \
    -lboost_program_options -lstdc++ \
    -o "$out_bin"
}

build_arm64_gen() {
  local main_src="$1"
  local out_bin="$2"
  "${TARGET_CXX_ARR[@]}" "${TARGET_CXXFLAGS_ARR[@]}" -std=c++17 \
    -I. -Itools/code_generators -Idependencies/pugixml/src \
    "$main_src" tools/code_generators/serialization_xml.cpp dependencies/pugixml/src/pugixml.cpp \
    -Wl,-rpath,/usr/lib/aarch64-linux-gnu -Wl,-rpath,/lib/aarch64-linux-gnu \
    -lboost_program_options \
    -o "$out_bin"
}

smoke_generator() {
  local exe=("$@")
  rm -f /tmp/host-tools-smoke/test.h /tmp/host-tools-smoke/test.cpp
  "${exe[@]}" -x game/state/gamestate_serialize.xml -h /tmp/host-tools-smoke/test.h -o /tmp/host-tools-smoke/test.cpp >/tmp/host-tools-smoke/stdout.log 2>/tmp/host-tools-smoke/stderr.log
  test -s /tmp/host-tools-smoke/test.h
  test -s /tmp/host-tools-smoke/test.cpp
}

probe_native_compiler() {
  local compiler="$1"
  cat > /tmp/native-probe/probe.cpp <<'EOF'
#include <iostream>
int main(){ std::cout << "ok\n"; return 0; }
EOF
  "$compiler" -x c++ -std=c++17 /tmp/native-probe/probe.cpp -lstdc++ -o /tmp/native-probe/probe.bin >>"$LAST_NATIVE_LOG" 2>&1 && /tmp/native-probe/probe.bin >>"$LAST_NATIVE_LOG" 2>&1
}

add_path() {
  local p="$1"
  [ -n "$p" ] || return 0
  [ -x "$p" ] || return 0
  if [ -z "${SEEN[$p]+x}" ]; then
    CANDIDATES+=("$p")
    SEEN[$p]=1
  fi
}

  declare -A QSEEN=()
  QCANDIDATES=()
  for name in qemu-aarch64 qemu-aarch64-static; do
    p=$(command -v "$name" 2>/dev/null || true)
    if [ -n "$p" ] && [ -z "${QSEEN[$p]+x}" ]; then QCANDIDATES+=("$p"); QSEEN[$p]=1; fi
  done
  shopt -s nullglob
  for p in /usr/bin/qemu-aarch64 /usr/bin/qemu-aarch64-static /usr/local/bin/qemu-aarch64 /usr/local/bin/qemu-aarch64-static /opt/*/bin/qemu-aarch64 /opt/*/bin/qemu-aarch64-static; do
    if [ -z "${QSEEN[$p]+x}" ]; then QCANDIDATES+=("$p"); QSEEN[$p]=1; fi
  done
  shopt -u nullglob
  for q in "${QCANDIDATES[@]}"; do
    [ -x "$q" ] || continue

       smoke_generator "$q" /tmp/host-tools/OpenApoc_LuaGamestateSupportGen.arm64; then
      QEMU_PREFIX=
      break

    for prefix in /usr/aarch64-linux-gnu /usr/aarch64-linux-gnu/libc /usr /; do

         smoke_generator "$q" -L "$prefix" /tmp/host-tools/OpenApoc_LuaGamestateSupportGen.arm64; then

        break 2
      fi
    done

  done

#!/bin/sh
exec "$FOUND_QEMU" -L "$QEMU_PREFIX" /tmp/host-tools/OpenApoc_GamestateSerializeGen.arm64 "\$@"
EOF
      cat > /tmp/host-tools/OpenApoc_LuaGamestateSupportGen <<EOF

exec "$FOUND_QEMU" -L "$QEMU_PREFIX" /tmp/host-tools/OpenApoc_LuaGamestateSupportGen.arm64 "\$@"
EOF

#!/bin/sh

EOF

#!/bin/sh
exec "$FOUND_QEMU" /tmp/host-tools/OpenApoc_LuaGamestateSupportGen.arm64 "\$@"

      chmod +x /tmp/host-tools/OpenApoc_GamestateSerializeGen /tmp/host-tools/OpenApoc_LuaGamestateSupportGen

    fi
  fi
fi

if [ "$HOST_TOOLS_READY" -eq 0 ]; then
  declare -A SEEN=()
  CANDIDATES=()
  for name in cc gcc clang c++ g++ clang++; do
    add_path "$(command -v "$name" 2>/dev/null || true)"
  done
  shopt -s nullglob
  for p in \
    /bin/cc /bin/c++ /bin/gcc /bin/g++ /bin/clang /bin/clang++ \
    /usr/bin/cc /usr/bin/c++ /usr/bin/gcc /usr/bin/g++ /usr/bin/clang /usr/bin/clang++ \
    /usr/bin/gcc-[0-9]* /usr/bin/g++-[0-9]* /usr/bin/clang-[0-9]* /usr/bin/clang++-[0-9]* \
    /usr/bin/x86_64-linux-gnu-gcc /usr/bin/x86_64-linux-gnu-g++ /usr/bin/x86_64-linux-gnu-gcc-[0-9]* /usr/bin/x86_64-linux-gnu-g++-[0-9]* \

    /usr/local/bin/gcc-[0-9]* /usr/local/bin/g++-[0-9]* /usr/local/bin/clang-[0-9]* /usr/local/bin/clang++-[0-9]* \
    /usr/lib/llvm-*/bin/clang /usr/lib/llvm-*/bin/clang++ \
    /opt/*/bin/cc /opt/*/bin/c++ /opt/*/bin/gcc /opt/*/bin/g++ /opt/*/bin/clang /opt/*/bin/clang++; do

  shopt -u nullglob

  for compiler in "${CANDIDATES[@]}"; do
    echo "$compiler" >> "$CANDIDATES_LOG"

         smoke_generator /tmp/host-tools/OpenApoc_LuaGamestateSupportGen.native; then
