make_copygen() {
  local src_h="$1"
  local src_cpp="$2"
  local out="$3"
  cat > "$out" <<EOF
#!/bin/sh
set -eu
SRC_H='$src_h'
SRC_CPP='$src_cpp'
OUT_H=
OUT_CPP=
while [ "\$#" -gt 0 ]; do
  case "\$1" in
    -x)
      shift 2
      ;;
    -h)
      OUT_H="\$2"
      shift 2
      ;;
    -o)
      OUT_CPP="\$2"
      shift 2
      ;;
    *)
      shift
      ;;
  esac
done
[ -n "\${OUT_H}" ]
[ -n "\${OUT_CPP}" ]
cp "\$SRC_H" "\$OUT_H"
cp "\$SRC_CPP" "\$OUT_CPP"
EOF
  chmod +x "$out"
}

build_native_gen() {
  local compiler="$1"
  local main_src="$2"
  local out_bin="$3"
  "${NATIVE_ENV_CLEAN[@]}" "$compiler" -x c++ -std=c++17 \
    "$HOST_SHIM_INC" -I. -Itools/code_generators -Idependencies/pugixml/src \
    "$main_src" tools/code_generators/serialization_xml.cpp dependencies/pugixml/src/pugixml.cpp \
    -lstdc++ \
    -o "$out_bin"
}

build_arm64_gen() {
  local main_src="$1"
  local out_bin="$2"
  "${TARGET_CXX_ARR[@]}" "${TARGET_CXXFLAGS_ARR[@]}" -std=c++17 \
    -I. -Itools/code_generators -Idependencies/pugixml/src \
    "$main_src" tools/code_generators/serialization_xml.cpp dependencies/pugixml/src/pugixml.cpp \
    -Wl,-rpath,/usr/lib/aarch64-linux-gnu -Wl,-rpath,/lib/aarch64-linux-gnu \
    -lboost_program_options \
    -o "$out_bin"
}

smoke_generator() {
  local exe=("$@")
  rm -f /tmp/host-tools-smoke/test.h /tmp/host-tools-smoke/test.cpp
  "${exe[@]}" -x game/state/gamestate_serialize.xml -h /tmp/host-tools-smoke/test.h -o /tmp/host-tools-smoke/test.cpp >/tmp/host-tools-smoke/stdout.log 2>/tmp/host-tools-smoke/stderr.log
  test -s /tmp/host-tools-smoke/test.h
  test -s /tmp/host-tools-smoke/test.cpp
}

probe_native_compiler() {
  local compiler="$1"
  cat > /tmp/native-probe/probe.cpp <<'EOF'
#include <iostream>
int main(){ std::cout << "ok\n"; return 0; }
EOF
  "${NATIVE_ENV_CLEAN[@]}" "$compiler" -x c++ -std=c++17 /tmp/native-probe/probe.cpp -lstdc++ -o /tmp/native-probe/probe.bin >>"$LAST_NATIVE_LOG" 2>&1 && /tmp/native-probe/probe.bin >>"$LAST_NATIVE_LOG" 2>&1
}

declare -A SEEN=()
CANDIDATES=()
add_path() {
  local p="$1"
  [ -n "$p" ] || return 0
  [ -x "$p" ] || return 0
  if [ -z "${SEEN[$p]+x}" ]; then
    CANDIDATES+=("$p")
    SEEN[$p]=1
  fi
}

add_path "${HOST_CXX:-}"
add_path "${CXX_FOR_BUILD:-}"
add_path "$(command -v g++ 2>/dev/null || true)"
add_path "/usr/bin/g++"
add_path "$(command -v c++ 2>/dev/null || true)"

for compiler in "${CANDIDATES[@]}"; do
  echo "$compiler" >> "$CANDIDATES_LOG"
  : > "$LAST_NATIVE_LOG"
  if probe_native_compiler "$compiler"; then
    FOUND_NATIVE_CXX="$compiler"
    if build_native_gen "$compiler" tools/code_generators/gamestate_serialize_gen/main.cpp /tmp/host-tools/OpenApoc_GamestateSerializeGen.native >>"$LAST_NATIVE_LOG" 2>&1 && \
       build_native_gen "$compiler" tools/code_generators/luagamestate_support_gen/main.cpp /tmp/host-tools/OpenApoc_LuaGamestateSupportGen.native >>"$LAST_NATIVE_LOG" 2>&1 && \
       smoke_generator /tmp/host-tools/OpenApoc_GamestateSerializeGen.native && \
       smoke_generator /tmp/host-tools/OpenApoc_LuaGamestateSupportGen.native; then
      ln -sf /tmp/host-tools/OpenApoc_GamestateSerializeGen.native /tmp/host-tools/OpenApoc_GamestateSerializeGen
      ln -sf /tmp/host-tools/OpenApoc_LuaGamestateSupportGen.native /tmp/host-tools/OpenApoc_LuaGamestateSupportGen
      HOST_TOOLS_READY=1
      break
    fi
  fi
done

  echo "--- native compiler log tail ---" >&2
  tail -n 150 "$LAST_NATIVE_LOG" >&2 || true
  exit 1
fi
